cmake_minimum_required(VERSION 3.13)
set(PICO_BOARD pico2_w CACHE STRING "Target board")
if(NOT PICO_SDK_PATH)
    message(FATAL_ERROR "Set PICO_SDK_PATH to Pico SDK 2.2.0")
endif()
include("${PICO_SDK_PATH}/external/pico_sdk_import.cmake")
project(multinodedaq_pico C CXX ASM)
set(CMAKE_C_STANDARD 11)
set(CMAKE_CXX_STANDARD 17)
pico_sdk_init()
execute_process(COMMAND git rev-parse --short=12 HEAD WORKING_DIRECTORY "${CMAKE_CURRENT_SOURCE_DIR}" OUTPUT_VARIABLE BUILD_COMMIT OUTPUT_STRIP_TRAILING_WHITESPACE)
execute_process(COMMAND git status --porcelain WORKING_DIRECTORY "${CMAKE_CURRENT_SOURCE_DIR}" OUTPUT_VARIABLE BUILD_DIRTY)
if(BUILD_DIRTY)
    string(APPEND BUILD_COMMIT "-dirty")
endif()
add_executable(multinodedaq_diagnostic diagnostic.c)
target_link_libraries(multinodedaq_diagnostic pico_stdlib pico_unique_id pico_cyw43_arch_none hardware_clocks)
target_compile_definitions(multinodedaq_diagnostic PRIVATE MND_BUILD="${BUILD_COMMIT}" PICO_STDIO_USB_STDOUT_TIMEOUT_US=10000)
target_compile_options(multinodedaq_diagnostic PRIVATE -Wall -Wextra -Werror)
pico_enable_stdio_usb(multinodedaq_diagnostic 1)
pico_enable_stdio_uart(multinodedaq_diagnostic 0)
pico_set_program_name(multinodedaq_diagnostic "MultiNodeDAQ diagnostic")
pico_set_program_version(multinodedaq_diagnostic "0.5.0-stage5-bringup")
pico_add_extra_outputs(multinodedaq_diagnostic)

if(MND_PRIVATE_DIR)
    add_executable(multinodedaq_stream stream.c vendor/cjson/cJSON.c)
    target_include_directories(multinodedaq_stream PRIVATE . "${MND_PRIVATE_DIR}")
    target_link_libraries(multinodedaq_stream pico_stdlib pico_unique_id pico_rand pico_cyw43_arch_lwip_poll)
    target_compile_definitions(multinodedaq_stream PRIVATE MND_BUILD="${BUILD_COMMIT}" CJSON_NESTING_LIMIT=16 PICO_STDIO_USB_STDOUT_TIMEOUT_US=1000)
    target_compile_options(multinodedaq_stream PRIVATE -Wall -Wextra)
    pico_enable_stdio_usb(multinodedaq_stream 1)
    pico_enable_stdio_uart(multinodedaq_stream 0)
    pico_set_program_name(multinodedaq_stream "MultiNodeDAQ counter stream")
    pico_set_program_version(multinodedaq_stream "0.5.0-stage5")
    pico_add_extra_outputs(multinodedaq_stream)
endif()
